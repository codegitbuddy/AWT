EmployeeManager.php
<?php 
require_once __DIR__ . "/Employee.php";

class EmployeeManager
{
	private string $fileName;
	
	public function __construct(string $fileName)
	{
		$this->fileName = $fileName;
		
		if(!file_exists($this->fileName)){
			file_put_contents($this->fileName, "[]");
		}
	}
	
	public function getAll(): array 
	{
		$json = file_get_contents($this->fileName);
		$employees = json_decode($json ?: "[]", true);
		return is_array($employees) ? $employees : [];
	}
	
	private function saveAll(array $employees): void 
	{
		file_put_contents($this->fileName, json_encode($employees, JSON_PRETTY_PRINT),LOCK_EX);
	}
	
	public function add(Employee $employee): bool
	{
		$employees = $this->getAll();
		
		foreach ($employees as $item) {
			if ((int) $item["id"] === $employee->id){
				return false;
			}
		}
		$employees[] = $employee->toArray();
		$this->saveAll($employees);
		return true;		
	}
	
	public function update(Employee $employee): bool
	{
		$employees=$this->getAll();
		
		foreach ($employees as $index => $item){
			if ((int) $item["id"] === $employee->id){
				$employees[$index] = $employee->toArray();
				$this->saveAll($employees);
				return true;
			}
		}
		return false;
	}
}
